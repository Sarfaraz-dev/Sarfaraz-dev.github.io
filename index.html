Q1: Difference between Procedural and Object-Oriented Programming
What is Procedural Programming? Compare it with object-oriented programming.
Procedural Programming:

Procedural programming is based on procedure calls. Programs are divided into functions or procedures that operate on data. Data and functions are separate.

  • Characteristics: Top-down approach, Global data, Functions manipulate data
  • Advantages: Easy to understand, Simple for small programs
  • Disadvantages: Difficult to maintain large programs, No data hiding
  • Example Languages: C, Pascal, FORTRAN
Object-Oriented Programming (OOP):

OOP treats data as objects that contain both data and functions. It follows bottom-up approach with concepts like encapsulation, inheritance, polymorphism.

  • Characteristics: Data hiding, Reusability, Modularity
  • Advantages: Easy maintenance, Code reusability, Better security
  • Disadvantages: Steeper learning curve, Overhead of objects
  • Example Languages: C++, Java, Python
Key Differences:
Aspect Procedural OOP
Approach Top-down Bottom-up
Data Access Global Encapsulated
Reusability Low High (Inheritance)
Security Low High (Data hiding)
Maintenance Difficult Easy
Q2: Check if Character is Vowel
Write a Program to check a character is vowel or not, using if else if ladder and switch case statement. 
// Check Vowel using If-Else Ladder AND Switch Case
#include <iostream>
#include <cctype>
using namespace std;

int main() {
    char ch;
    int choice;

    cout << "Enter a character: ";
    cin >> ch;
    
    // Converting to uppercase for easier checking
    char upperChar = toupper(ch);

    cout << "\nChoose Method:\n1. If-Else Ladder\n2. Switch Case\nEnter choice: ";
    cin >> choice;

    if (choice == 1) {
        // Method 1: If-Else Ladder
        if (upperChar == 'A' || upperChar == 'E' || upperChar == 'I' || upperChar == 'O' || upperChar == 'U')
            cout << ch << " is a Vowel (checked via If-Else)." << endl;
        else if (isalpha(ch))
            cout << ch << " is a Consonant (checked via If-Else)." << endl;
        else
            cout << ch << " is not an alphabet." << endl;
    } 
    else {
        // Method 2: Switch Case
        if (isalpha(ch)) {
            switch (upperChar) {
                case 'A':
                case 'E':
                case 'I':
                case 'O':
                case 'U':
                    cout << ch << " is a Vowel (checked via Switch)." << endl;
                    break;
                default:
                    cout << ch << " is a Consonant (checked via Switch)." << endl;
            }
        } else {
            cout << ch << " is not an alphabet." << endl;
        }
    }
    return 0;
}
Output:
Enter a character: E Choose Method: 1. If-Else Ladder 2. Switch Case Enter choice: 2 E is a Vowel (checked via Switch).

Q4: Sum of First N Natural Numbers
Write a program using all types of loops to print the sum of the first n natural numbers. 
// Sum of first n natural numbers using ALL Loops
#include <iostream>
using namespace std;

int main() {
    int n, sum = 0, i;
    cout << "Enter value of n: ";
    cin >> n;

    // 1. Using FOR Loop
    sum = 0;
    for(i = 1; i <= n; i++) {
        sum += i;
    }
    cout << "Sum using For Loop: " << sum << endl;

    // 2. Using WHILE Loop
    sum = 0;
    i = 1;
    while(i <= n) {
        sum += i;
        i++;
    }
    cout << "Sum using While Loop: " << sum << endl;

    // 3. Using DO-WHILE Loop
    sum = 0;
    i = 1;
    do {
        sum += i;
        i++;
    } while(i <= n);
    cout << "Sum using Do-While Loop: " << sum << endl;

    return 0;
}

Q5: Sum of Diagonal Elements of Matrix
Write a Program to create a matrix of 3 x 3 and find the sum of left diagonal as well as right diagonal. 
// Sum of Left and Right Diagonals of 3x3 Matrix
#include <iostream>
using namespace std;

int main() {
    int A[3][3];
    int i, j, leftSum = 0, rightSum = 0;

    cout << "Enter elements of 3x3 Matrix:\n";
    for(i = 0; i < 3; i++) {
        for(j = 0; j < 3; j++) {
            cin >> A[i][j];
        }
    }

    // Calculating Sums
    for(i = 0; i < 3; i++) {
        for(j = 0; j < 3; j++) {
            // Left Diagonal Condition (i == j)
            if(i == j) {
                leftSum += A[i][j];
            }
            // Right Diagonal Condition (i + j == size - 1)
            if((i + j) == 2) {
                rightSum += A[i][j];
            }
        }
    }

    cout << "\nMatrix:\n";
    for(i = 0; i < 3; i++) {
        for(j = 0; j < 3; j++) {
            cout << A[i][j] << " ";
        }
        cout << endl;
    }

    cout << "\nSum of Left Diagonal: " << leftSum << endl;
    cout << "Sum of Right Diagonal: " << rightSum << endl;

    return 0;
}
Output:
Enter elements of 3x3 Matrix: 1 2 3 4 5 6 7 8 9 Matrix: 1 2 3 4 5 6 7 8 9 Sum of Left Diagonal: 15 Sum of Right Diagonal: 15

Q7: Check Armstrong Number
Write a program to check whether a number is Armstrong or not using a function with and without arguments. 
// Armstrong Check (Function Overloading)
#include <iostream>
#include <cmath>
using namespace std;

// Function WITHOUT arguments
void checkArmstrong() {
    int n, original, remainder, result = 0, digits = 0;
    cout << "Enter an integer: ";
    cin >> n;
    original = n;

    int temp = n;
    while (temp != 0) { temp /= 10; digits++; }

    temp = n;
    while (temp != 0) {
        remainder = temp % 10;
        result += pow(remainder, digits);
        temp /= 10;
    }

    if (result == original)
        cout << original << " is an Armstrong number." << endl;
    else
        cout << original << " is not an Armstrong number." << endl;
}

// Function WITH arguments
void checkArmstrong(int n) {
    int original = n, remainder, result = 0, digits = 0;
    int temp = n;
    while (temp != 0) { temp /= 10; digits++; }

    temp = n;
    while (temp != 0) {
        remainder = temp % 10;
        result += pow(remainder, digits);
        temp /= 10;
    }

    if (result == original)
        cout << original << " is an Armstrong number (Checked via Argument)." << endl;
    else
        cout << original << " is not an Armstrong number (Checked via Argument)." << endl;
}

int main() {
    int choice, num;
    cout << "1. Check without argument\n2. Check with argument\nEnter choice: ";
    cin >> choice;

    if(choice == 1) {
        checkArmstrong();
    } else {
        cout << "Enter number to check: ";
        cin >> num;
        checkArmstrong(num);
    }
    return 0;
}

Q10: Fibonacci Series using Class and Object
Write a Program to generate Fibonacci series using class and object. 
// Fibonacci Series using Class
#include <iostream>
using namespace std;

class Fibonacci {
private:
    int n, first, second, next;
    
public:
    // Constructor to initialize values
    Fibonacci() {
        first = 0;
        second = 1;
    }
    
    // Function to input number of terms
    void input() {
        cout << "Enter number of terms: ";
        cin >> n;
    }
    
    // Function to generate and display Fibonacci series
    void generate() {
        cout << "\nFibonacci Series: " << endl;
        
        if(n >= 1)
            cout << first << " ";
        if(n >= 2)
            cout << second << " ";
            
        for(int i = 3; i <= n; i++) {
            next = first + second;
            cout << next << " ";
            first = second;
            second = next;
        }
        cout << endl;
    }
};

int main() {
    Fibonacci fib;  // Create object
    
    fib.input();     // Input number of terms
    fib.generate();  // Generate Fibonacci series
    
    return 0;
}
Q11: Stack - Push and Pop Operations
What is Stack? Write Algorithm of Push and Pop operation.
What is Stack?

A Stack is a linear data structure that follows the LIFO (Last In First Out) principle. The element inserted last is the first one to be removed.

Algorithm for PUSH Operation:
Algorithm PUSH(STACK, TOP, MAX, ITEM):

Step 1: START
Step 2: IF TOP = MAX - 1 THEN
            PRINT "Stack Overflow"
            GOTO Step 6
        END IF
Step 3: TOP = TOP + 1
Step 4: STACK[TOP] = ITEM
Step 5: PRINT "Element inserted successfully"
Step 6: STOP
Algorithm for POP Operation:
Algorithm POP(STACK, TOP, ITEM):

Step 1: START
Step 2: IF TOP = -1 THEN
            PRINT "Stack Underflow"
            GOTO Step 6
        END IF
Step 3: ITEM = STACK[TOP]
Step 4: TOP = TOP - 1
Step 5: PRINT "Element deleted:", ITEM
Step 6: STOP
Q12: Queue - Insert and Delete Operations
What is Queue? Write an Algorithm of insert and delete Operation.
What is Queue?

A Queue is a linear data structure that follows the FIFO (First In First Out) principle. The element inserted first is the first one to be removed.

Algorithm for INSERT (Enqueue) Operation:
Algorithm INSERT(QUEUE, FRONT, REAR, MAX, ITEM):

Step 1: START
Step 2: IF REAR = MAX - 1 THEN
            PRINT "Queue Overflow"
            GOTO Step 7
        END IF
Step 3: IF FRONT = -1 THEN
            FRONT = 0
        END IF
Step 4: REAR = REAR + 1
Step 5: QUEUE[REAR] = ITEM
Step 6: PRINT "Element inserted successfully"
Step 7: STOP
Algorithm for DELETE (Dequeue) Operation:
Algorithm DELETE(QUEUE, FRONT, REAR, ITEM):

Step 1: START
Step 2: IF FRONT = -1 OR FRONT > REAR THEN
            PRINT "Queue Underflow"
            GOTO Step 7
        END IF
Step 3: ITEM = QUEUE[FRONT]
Step 4: FRONT = FRONT + 1
Step 5: IF FRONT > REAR THEN
            FRONT = -1
            REAR = -1
        END IF
Step 6: PRINT "Element deleted:", ITEM
Step 7: STOP
Q13: Bubble Sort Program and Algorithm
Write a Program along with an Algorithm of Bubble Sort.
Algorithm for Bubble Sort:
Algorithm BUBBLE_SORT(ARRAY, N):

Step 1: START
Step 2: FOR i = 0 TO N-2
            FOR j = 0 TO N-i-2
                IF ARRAY[j] > ARRAY[j+1] THEN
                    SWAP ARRAY[j] and ARRAY[j+1]
                END IF
            END FOR
        END FOR
Step 3: PRINT "Sorted Array"
Step 4: STOP
Program:
// Bubble Sort Implementation
#include <iostream>
using namespace std;

void bubbleSort(int arr[], int n) {
    int temp;
    
    for(int i = 0; i < n-1; i++) {
        // Last i elements are already sorted
        bool swapped = false;
        for(int j = 0; j < n-i-1; j++) {
            // Swap if element is greater than next
            if(arr[j] > arr[j+1]) {
                temp = arr[j];
                arr[j] = arr[j+1];
                arr[j+1] = temp;
                swapped = true;
            }
        }
        // If no swapping, array is sorted
        if(!swapped) break;
    }
}

int main() {
    int arr[] = {64, 34, 25, 12, 22};
    int n = sizeof(arr)/sizeof(arr[0]);
    
    cout << "Original array: ";
    for(int i = 0; i < n; i++) {
        cout << arr[i] << " ";
    }
    cout << endl;
    
    bubbleSort(arr, n);
    
    cout << "Sorted array: ";
    for(int i = 0; i < n; i++) {
        cout << arr[i] << " ";
    }
    cout << endl;
    
    return 0;
}
Q14: Binary Search Algorithm and Program
Write a program along with an Algorithm for Binary Search on sorted array.
Algorithm for Binary Search:
Algorithm BINARY_SEARCH(ARR, N, VAL):

Step 1: START
Step 2: Set LOW = 0, HIGH = N - 1
Step 3: Repeat Step 4 while LOW <= HIGH
Step 4:      Set MID = (LOW + HIGH) / 2
            IF ARR[MID] = VAL THEN
                PRINT "Value found at index", MID
                EXIT
            ELSE IF ARR[MID] < VAL THEN
                Set LOW = MID + 1
            ELSE
                Set HIGH = MID - 1
            END IF
Step 5: PRINT "Value not found"
Step 6: STOP
Program:
// Binary Search Implementation
#include <iostream>
using namespace std;

int binarySearch(int arr[], int n, int key) {
    int low = 0, high = n - 1;
    int mid;
    
    while(low <= high) {
        mid = (low + high) / 2;
        
        if(arr[mid] == key) {
            return mid;
        } else if(arr[mid] < key) {
            low = mid + 1;
        } else {
            high = mid - 1;
        }
    }
    return -1;
}

int main() {
    int arr[10], n = 10, key;
    
    cout << "Enter 10 sorted numbers:" << endl;
    for(int i = 0; i < n; i++) {
        cout << "Element [" << i << "]: ";
        cin >> arr[i];
    }
    
    cout << "\nEnter number to search: ";
    cin >> key;
    
    int result = binarySearch(arr, n, key);
    
    if(result != -1) {
        cout << "Element found at index: " << result << endl;
    } else {
        cout << "Element not found in array" << endl;
    }
    
    return 0;
}
